/*
  SPIKE: /_image with an SVG exemption layered over Cloudflare's default.

  Cloudflare Images only accepts jpeg, png, gif, webp and avif, so anything SVG
  has to be served without going through it. Everything else is handed to the
  adapter's own endpoint unchanged.
*/
import type { APIContext } from "astro";

import { GET as passthroughFromAssets } from "@astrojs/cloudflare/image-passthrough-endpoint";
import { GET as transformWithCloudflare } from "@astrojs/cloudflare/image-transform-endpoint";

export const prerender = false;

// In dev the source lives behind Vite (`/@fs/...`), not the ASSETS binding, so
// the built passthrough endpoint can't reach it.
const serveSvgFromDevServer = async (request: Request) => {
  const url = new URL(request.url);
  const href = url.searchParams.get("href");
  if (!href) {
    return new Response("Bad Request", { status: 400 });
  }
  const source = new URL(href, url.origin);
  if (source.origin !== url.origin) {
    return new Response("Forbidden", { status: 403 });
  }
  const response = await fetch(source);
  if (!response.ok) {
    return new Response("Not Found", { status: 404 });
  }
  return new Response(response.body, {
    headers: {
      "Cache-Control": "public, max-age=31536000",
      "Content-Type": "image/svg+xml",
    },
    status: 200,
  });
};

export const GET = (context: APIContext) => {
  const isSvg =
    new URL(context.request.url).searchParams.get("f") === "svg";

  if (!isSvg) {
    return transformWithCloudflare(context);
  }

  return import.meta.env.DEV
    ? serveSvgFromDevServer(context.request)
    : passthroughFromAssets(context);
};
